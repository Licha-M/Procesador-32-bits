; =======================================================================
; PRUEBA 8: Paginación (MMU), Page Walk Puro, Permisos y Protección
; =======================================================================

JMP main

; === EXCEPTION HANDLER (Dirección fija en ROM: 0xFFF0000C) ===
handler_mmu:
    CYE R7, R10        ; Lee causa de la excepción (RE 7). Debe ser 1 (Page Fault).
    CYE R6, R11        ; Lee CR0 (RE 6). La parte alta contiene el Error Code.
    HLT                ; Detiene el procesador. Prueba completada con éxito.

; === PROGRAMA PRINCIPAL ===
main:
    ; 1. Configurar el vector SysJMP (Apunta a handler_mmu en 0xFFF0000C)
    H LDI R1, 0xFFF0
    SLT ADI R1, 0x000C
    CYR R1, R12        ; RE 12 = SysJMP

    ; 2. Inicialización de las Tablas de Páginas en RAM

    ; --- Page Directory (PD) en 0x1000 ---
    H LDI R1, 0x0000
    SLT ADI R1, 0x1000      ; R1 = Base PD (0x1000)

    ; PD[0] -> PT_RAM (0x2000). Flags: Present(1) = 0x2001
    H LDI R2, 0x0000
    SLT ADI R2, 0x2001
    int STR R1, R2, 0       ; Escribe 0x2001 en 0x1000

    ; PD[1023] -> PT_ROM (0x3000). Flags: Present(1) + Executable(16) = 0x3011
    H LDI R2, 0x0000
    SLT ADI R2, 0x3011
    int STR R1, R2, 4092    ; Escribe 0x3011 en 0x1FFC (1023 * 4)

    ; --- Page Table de ROM (PT_ROM) en 0x3000 ---
    H LDI R1, 0x0000
    SLT ADI R1, 0x3000      ; R1 = Base PT_ROM (0x3000)

    ; PTE[768] Mapeo de Virtual 0xFFF00000 -> Física 0xFFF00000 (ROM)
    ; Flags: Present(1) + Executable(16) = 0x0011. Valor: 0xFFF00011
    ; Dirección final de la entrada en RAM: 0x3000 + 3072 = 0x3C00
    H LDI R2, 0xFFF0
    SLT ADI R2, 0x0011
    int STR R1, R2, 3072    ; Escribe 0xFFF00011 en 0x3C00

    ; --- Page Table de RAM (PT_RAM) en 0x2000 ---
    H LDI R1, 0x0000
    SLT ADI R1, 0x2000      ; R1 = Base PT_RAM (0x2000)

    ; PTE[5] Página de Lectura/Escritura Válida (0x5000 -> 0x5000)
    ; Flags: Present(1) = 0x5001. Offset: 5 * 4 = 20
    H LDI R2, 0x0000
    SLT ADI R2, 0x5001
    int STR R1, R2, 20      ; Escribe 0x5001 en 0x2014

    ; PTE[6] Página de Solo Lectura (0x6000 -> 0x6000)
    ; Flags: Present(1) + ReadOnly(2) = 0x6003. Offset: 6 * 4 = 24
    H LDI R2, 0x0000
    SLT ADI R2, 0x6003
    int STR R1, R2, 24      ; Escribe 0x6003 en 0x2018

    ; 3. Cargar Registros de Control de la MMU
    H LDI R1, 0x0000
    SLT ADI R1, 0x1000
    CYR R1, R4         ; RE 4 (CR3) = Base del Page Directory (0x1000)

    LDI R1, 1
    CYR R1, R3         ; RE 3 (PCID) = 1

    ; 4. Activar la MMU (Sin TLB para evaluar Page Walk puro)
    ; CR0 Bit 0 (En_MMU) = 1, Bit 1 (En_TLB) = 0
    LDI R1, 3
    CYR R1, R6         ; RE 6 (CR0) = 1. ¡Memoria Virtual Activa!

    ; 5. Prueba 1: Acceso a Memoria Válido (0x5000)
    H LDI R1, 0x0000
    SLT ADI R1, 0x5000
    LDI R2, 0x9999
    int STR R1, R2, 0       ; MMU traduce 0x5000 y permite la escritura

    ; 6. Prueba 2: Forzar Excepción de Page Fault (0x6000)
    H LDI R1, 0x0000
    SLT ADI R1, 0x6000
    LDI R2, 0xDEAD
    int STR R1, R2, 0       ; Al ser ReadOnly (Bit 1=1), la MMU aborta
                           ; y salta inmediatamente a handler_mmu (0xFFF0000C).

    HLT                ; Trampa de seguridad (no debe alcanzarse).
