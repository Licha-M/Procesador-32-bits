; ==========================================================
; Programa 2: Suma de Arreglo en RAM
; Escribe valores en RAM, los lee con un puntero y los suma.
; ==========================================================

; 1. Escribir datos de prueba en la RAM
; Usamos R1 como nuestra dirección base (ej. dirección 100)
LDI R1, 100

LDI R2, 15
int STR R1, R2, 0   ; RAM[100 + 0] = 15

LDI R2, 25
int STR R1, R2, 4   ; RAM[100 + 4] = 25 (Asumiendo 4 bytes por palabra)

LDI R2, 10
int STR R1, R2, 8   ; RAM[100 + 8] = 10

; 2. Leer y sumar los datos
LDI R3, 0           ; R3 será nuestro acumulador (Suma total)
LDI R4, 3           ; R4 es el contador de elementos del arreglo

loop_suma:
; Cargamos en R5 el valor apuntado por la dirección en R1 con offset 0
int LOD R1, R5, 0   
ADD R3, R5, R3      ; Acumulamos: R3 = R3 + R5

; Avanzamos el puntero de memoria en 4 bytes (1 palabra de 32 bits)
ADI R1, 4           

; Decrementamos el contador y saltamos si no es cero
ADI R4, -1          
BRH !=, loop_suma   

; Al terminar, R3 debería contener 50
HLT
