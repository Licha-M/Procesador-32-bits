; ============================================================
;  Programa de ejemplo — Procesador 32 bits
;
;  NOVEDADES:
;   - Etiquetas:  declarar con "nombre:" y usar directamente
;                 en JMP, BRH y CAL. El ensamblador genera
;                 automáticamente la secuencia LDI+LSH+ADD.
;   - R14 y R15 son registros reservados para los saltos con
;     etiqueta. No uses R14/R15 para otros datos.
;   - Condiciones de BRH: =  !=  >  <  >=  <=  OV  NOV
; ============================================================


; ─── Inicio del programa ─────────────────────────────────
inicio:
    NOP                         ; Ciclo vacío al arrancar


; ─── Carga de valores ────────────────────────────────────
    LDI R0, 0                   ; R0 = 0  (contador del bucle)
    LDI R1, 10                  ; R1 = 10 (límite)
    LDI R2, 1                   ; R2 = 1  (incremento)


; ─── Bucle principal (cuenta de 0 a 9) ──────────────────
bucle:
    ADD R0, R2, R0              ; R0 = R0 + 1
    SUB R0, R1, R3              ; R3 = R0 - límite (para comparar)
    BRH <, bucle                ; Si R0 < R1, volver a 'bucle'
                                ; ↑ El ensamblador genera:
                                ;   LDI  R15, 0x0000   (high16 de addr bucle)
                                ;   LDI  R14, 16
                                ;   LSH  R15, R14, R15
                                ;   LDI  R14, 0x0003   (low16 de addr bucle)
                                ;   ADD  R15, R14, R15
                                ;   BRH  <, R15


; ─── Llamada a subrutina con etiqueta ────────────────────
    CAL calcular                ; Llama a la subrutina 'calcular'


; ─── Salto incondicional con etiqueta ────────────────────
    JMP fin                     ; Ir al final del programa


; ─── Subrutina: calcular ─────────────────────────────────
calcular:
    MUL R0, R1, R3              ; R3 = R0 * R1
    DIV R3, R2, R4              ; R4 = R3 / R2
    AND R0, R1, R5              ; R5 = R0 AND R1
    XOR R0, R1, R6              ; R6 = R0 XOR R1
    RET                         ; Volver al llamador


; ─── Acceso a memoria RAM ────────────────────────────────
mem_test:
    LDI R0, 0x0100              ; R0 = dirección base
    int   LOD R0, R7, 0         ; R7 = int en RAM[R0 + 0]
    short LOD R0, R8, 4         ; R8 = short en RAM[R0 + 4]
    char  LOD R0, R9, 6         ; R9 = byte en RAM[R0 + 6]

    ADI R7, 1                   ; R7++
    int   STR R0, R7, 0         ; Guardar R7 de vuelta en RAM
    char  STR R0, R9, 6         ; Guardar byte


; ─── Uso de todas las condiciones de BRH ────────────────
condiciones_demo:
    LDI R0, 5
    LDI R1, 5
    SUB R0, R1, R3              ; R3 = 0 → activa Zero flag

    BRH =,   fin_cond           ; Salta si R0 == R1 (Zero activo)
    BRH !=,  fin_cond           ; Salta si R0 != R1
    BRH >,   fin_cond           ; Salta si R0 > R1  (con signo)
    BRH <,   fin_cond           ; Salta si R0 < R1  (con signo)
    BRH >=,  fin_cond           ; Salta si R0 >= R1
    BRH <=,  fin_cond           ; Salta si R0 <= R1
    BRH OV,  fin_cond           ; Salta si hay overflow
    BRH NOV, fin_cond           ; Salta si NO hay overflow

fin_cond:
    NOP


; ─── Overflow / Carry ────────────────────────────────────
    LDI R0, 0xFFFF              ; R0 = 65535 (máximo 16 bits)
    LDI R1, 1
    ADD R0, R1, R2              ; R2 = overflow!
    GOF R3                      ; R3 = bit de overflow de la op anterior


; ─── Registros especiales ────────────────────────────────
    CYE R0, R1                  ; R1 ← registro especial R0
    CYR R2                      ; Registro especial ← R2


; ─── Sistema Operativo ────────────────────────────────────
    SCL                         ; System call
    SRT                         ; Retorno del SO


; ─── Fin ─────────────────────────────────────────────────
fin:
    HLT                         ; Detiene el procesador
